def candidate(kind):
    source = inspect.getsource(dictionary.get_words)
    if kind == 'current':
        return dictionary.get_words, source
    expression = {'matches': 'MIN(id) AS first_id, MAX(id) AS last_id',
                  'scope': 'ANY_VALUE(bounds.first_id) AS first_id, ANY_VALUE(bounds.last_id) AS last_id'}[kind]
    source = source.replace("        cursor.execute('SELECT COUNT(*) AS total FROM ' + source + ' WHERE ' + where, params)\n        pagination = query.metadata(cursor.fetchone()['total'])", f"        aggregates = 'COUNT(*) AS total' + (', {expression}' if query.q else '')\n        cursor.execute('SELECT ' + aggregates + ' FROM ' + source + ' WHERE ' + where, params)\n        counted = cursor.fetchone()\n        pagination = query.metadata(counted['total'])")
    source = source.replace("        # Paginate IDs before reading the public text fields.", "        # Paginate IDs before reading the public text fields.")
    anchor="        cursor.execute(\n            'SELECT w.id, w.dictid, w.english, w.chinese '"
    assert source.count(anchor)==1
    source=source.replace(anchor,"        if query.q:\n            source = 'word'\n            where += ' AND id BETWEEN %s AND %s'\n            params = [*params[1:], counted['first_id'], counted['last_id']]\n"+anchor)
    assert 'counted = ' in source
    namespace=dictionary.__dict__.copy()
    exec(compile(source,'<candidate-'+kind+'>','exec'),namespace)
    return namespace['get_words'], source


@pytest.mark.parametrize('profile', list(DISTRIBUTIONS))
def test_candidates(mysql_app, profile):
    seed(mysql_app,'medium',profile)
    report={'environment':environment(mysql_app),'profile':profile,'size':'medium','candidates':{}}
    references={}
    for kind in ('current','matches','scope','current_repeat'):
        operation, source = candidate('current' if kind=='current_repeat' else kind)
        with patch.object(dictionary,'get_words',operation):
            results={}
            for name, action in cases():
                if (name.startswith(('words.','small_words.')) and name not in ('words.first','words.last','small_words.first','small_words.last')):
                    measured=measure(mysql_app,action,5)
                    if name in references: assert references[name]==measured['result_sha256']
                    else: references[name]=measured['result_sha256']
                    results[name]=measured
                    print(profile,kind,name,round(measured['service_ms']['median'],2),flush=True)
        report['candidates'][kind]={'source':source,'cases':results}
    save_report(Path(__file__).parent/f'candidates-{profile}.json',report)
