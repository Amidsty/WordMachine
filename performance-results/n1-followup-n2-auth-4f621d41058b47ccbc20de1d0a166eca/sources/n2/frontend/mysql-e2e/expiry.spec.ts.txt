import { api, expect, login, logout, rejected, test } from './support'

test('natural session expiry rejects stale forms, closes the old feed and permits deliberate login', async ({
  newActor,
}) => {
  const page = await newActor()
  await login(page, 'user1')
  const original = (await api(page, '/api/session')).body
  const oldCookie = (await page.context().cookies()).find((cookie) => cookie.name === 'wm_session')!
  await page.getByRole('link', { name: '修改资料', exact: true }).click()
  await page.getByLabel('个人简介', { exact: true }).fill('expired draft must not be saved')
  const live = await page.context().newPage()
  let opened = 0
  let closed = 0
  live.on('websocket', (socket) => {
    opened++
    socket.on('close', () => closed++)
  })
  await live.goto('/game/1/')
  await live.getByLabel('自动刷新（5 秒）', { exact: true }).uncheck()
  await expect(live.getByText('实时连接已订阅', { exact: false })).toBeVisible()
  await expect.poll(() => opened).toBeGreaterThan(0)

  // The harness sets a short lifetime only for this disposable app. Do not change
  // timestamps, the browser clock, server responses or the stored session.
  await expect
    .poll(async () => (await api(page, '/api/session')).body.authenticated, {
      timeout: 20000,
      intervals: [250],
    })
    .toBe(false)
  let writes = 0
  page.on('request', (request) => {
    if (request.method() === 'POST' && new URL(request.url()).pathname === '/api/user/1/update')
      writes++
  })
  const denied = page.waitForResponse(
    (response) =>
      new URL(response.url()).pathname === '/api/user/1/update' &&
      response.request().method() === 'POST',
  )
  await page.getByRole('button', { name: '保存修改', exact: true }).click()
  expect((await denied).status()).toBe(401)
  await expect(page).toHaveURL(/\/login\/\?next=/)
  await live.getByRole('button', { name: '刷新对局', exact: true }).click()
  await expect(live).toHaveURL(/\/login\/\?next=/)
  await expect.poll(() => closed).toBe(opened)
  expect(writes).toBe(1)

  const replay = await newActor()
  await replay.goto('/login/')
  // Retain the original token but bypass browser-side expiry as an adversarial
  // replay: the server must still reject it using its stored deadline.
  await replay.context().addCookies([{ ...oldCookie, expires: -1 }])
  await rejected(replay, '/api/game/1', 'GET', undefined, 401, 'unauthenticated')
  const anonymous = (await api(replay, '/api/session')).body
  expect(anonymous.authenticated).toBe(false)
  expect(anonymous.csrf_token).not.toBe(original.csrf_token)

  await login(page, 'user1')
  const renewed = (await api(page, '/api/session')).body
  expect(renewed.csrf_token).not.toBe(original.csrf_token)
  const newCookie = (await page.context().cookies()).find((cookie) => cookie.name === 'wm_session')!
  expect(newCookie.value).not.toBe(oldCookie.value)
  expect((await api(page, '/api/user/1')).body.user.introduction).toBeNull()
  await page.getByRole('link', { name: '修改资料', exact: true }).click()
  await expect(page.getByLabel('个人简介', { exact: true })).toHaveValue('')
  expect(writes).toBe(1) // Reauthentication does not replay the expired draft.
  await logout(page)
})
