import json
from datetime import datetime, timedelta

import click
import pymysql
from flask import current_app
from flask.cli import with_appcontext

from .db import transaction
from .services.auth import create_user
from .services.errors import ApiError


@click.command('create-admin')
@click.argument('username')
@with_appcontext
def create_admin_command(username):
    """Create an administrator after migration; password is entered interactively."""
    password = click.prompt('Password', hide_input=True, confirmation_prompt=True)
    try:
        user = create_user(username, password, role='root')
    except ApiError as error:
        raise click.ClickException(error.message) from None
    click.echo(f'Administrator created (id={user["id"]})')


def _conversion_report(convert, game_id, apply, expected_version):
    try:
        report = convert(game_id, apply=apply, expected_version=expected_version)
    except ApiError as error:
        click.echo(json.dumps({'game_id': game_id, 'code': error.code, 'message': error.message}))
        raise click.exceptions.Exit(1) from None
    except pymysql.MySQLError:
        # A lost commit acknowledgement is not evidence of rollback. Do not
        # print SQL/data/connection details or automatically retry the write.
        click.echo(
            json.dumps(
                {
                    'game_id': game_id,
                    'code': 'conversion_database_error',
                    'message': 'Database operation failed; inspect current state before retrying conversion',
                }
            )
        )
        raise click.exceptions.Exit(1) from None
    click.echo(json.dumps(report))


@click.command('convert-game-participants')
@click.option('--game-id', required=True, type=click.IntRange(1, 2147483647))
@click.option(
    '--apply',
    is_flag=True,
    help='Apply only on an authorized, backed-up database after reviewing a dry run.',
)
@click.option(
    '--expected-version',
    type=click.IntRange(0, 2147483647),
    help='Required with --apply; use the reviewed version.',
)
@with_appcontext
def convert_game_participants_command(game_id, apply, expected_version):
    """Inspect one game's membership by default; never convert questions/answers/ratings."""
    from .services.participant_conversion import convert_game

    _conversion_report(convert_game, game_id, apply, expected_version)

@click.option('--game-id', required=True, type=click.IntRange(1, 2147483647))

@click.command('purge-auth')
@with_appcontext
def purge_auth_command():
    """Remove expired sessions and inactive rate-limit buckets only."""
    now = datetime.utcnow()
    cutoff = now - timedelta(seconds=current_app.config['AUTH_LIMIT_WINDOW'])
    with transaction() as cursor:
        cursor.execute('DELETE FROM auth_session WHERE expires_at <= %s', (now,))
        sessions = cursor.rowcount
        cursor.execute('DELETE FROM auth_rate_limit WHERE window_started < %s', (cutoff,))
        buckets = cursor.rowcount
    click.echo(f'Removed {sessions} expired sessions and {buckets} inactive limit buckets')
