def parse_csv(content, *, max_bytes, max_rows):
    if not isinstance(content, str):
        raise ApiError('CSV must be a string')
    try:
        size = len(content.encode('utf-8'))
    except UnicodeEncodeError:
        raise ApiError('CSV contains invalid Unicode') from None
    if size > max_bytes:
        raise ApiError('CSV exceeds the size limit', 413, 'csv_too_large')
    reader = csv.reader(io.StringIO(content.lstrip('\ufeff'), newline=''), strict=True)
    rows, errors = [], []
    total = 0
    while True:
        line = reader.line_num + 1
        try:
            fields = next(reader)
        except StopIteration:
            break
        except csv.Error:
            errors.append({'line': line, 'message': 'Malformed CSV quoting or oversized field'})
            break
        if not fields or (len(fields) == 1 and not fields[0].strip()):
            continue
        total += 1
        if total > max_rows:
            raise ApiError(
                'CSV exceeds the row limit',
                413,
                'csv_too_many_rows',
                {'line': line, 'limit': max_rows},
            )
        try:
            if len(fields) != 2:
                raise ApiError('Expected exactly two columns: english,chinese (no header)')
            english = string(fields[0], 'English', minimum=1)
            chinese = string(fields[1], 'Chinese', minimum=1)
            rows.append((english, chinese))
        except ApiError as error:
            errors.append({'line': line, 'message': error.message})
        if len(errors) >= 50:
            break
    if errors:
        raise ApiError(
            'CSV validation failed; no words imported',
            400,
            'invalid_csv',
            {'errors': errors, 'limit': 50},
        )
    if not rows:
        raise ApiError('CSV contains no words', 400, 'invalid_csv')
    return rows


def csv_content(words):
    output = io.StringIO(newline='')
    writer = csv.writer(output, lineterminator='\r\n')
    for word in words:
        writer.writerow((word['english'], word['chinese']))
    return output.getvalue()


def _dictionary(cursor, dict_id, *, lock=False):
    cursor.execute(
        'SELECT id, dictname, ownerid FROM dict WHERE id = %s AND deleted = 0'
        + (' FOR UPDATE' if lock else ''),
        (dict_id,),
    )
    dictionary = cursor.fetchone()
    if not dictionary:
        raise ApiError('Dictionary not found', 404, 'not_found')
    return dictionary


def _writable_dictionary(cursor, actor_id, dict_id):
    actor = lock_actor(cursor, actor_id)
    dictionary = _dictionary(cursor, dict_id, lock=True)
    require_dictionary_owner(actor, dictionary)
    return dictionary


def list_dictionaries(actor, query=ListQuery()):
    where, params = 'd.deleted = 0', []
    if query.q:
        where += " AND d.dictname LIKE %s ESCAPE '!'"
        params.append(query.like)
    with reading() as cursor:
        cursor.execute('SELECT COUNT(*) AS total FROM dict d WHERE ' + where, params)
        pagination = query.metadata(cursor.fetchone()['total'])
        cursor.execute(
            'SELECT d.id, d.dictname, d.ownerid FROM dict d WHERE '
            + where
            + ' ORDER BY d.id DESC LIMIT %s OFFSET %s',
            [*params, query.per_page, page_offset(pagination)],
        )
        dictionaries = cursor.fetchall()
        # Aggregate only the accepted page, once. A correlated count can rescan the
        # whole word table for every dictionary when the legacy schema lacks an index.
        # Keep all three reads in this REPEATABLE READ snapshot.
        counts = {}
        if dictionaries:
            ids = [dictionary['id'] for dictionary in dictionaries]
            placeholders = ', '.join(['%s'] * len(ids))
            cursor.execute(
                'SELECT dictid, COUNT(*) AS word_count FROM word '
                'WHERE deleted = 0 AND dictid IN (' + placeholders + ') GROUP BY dictid',
                ids,
            )
            counts = {row['dictid']: row['word_count'] for row in cursor.fetchall()}
    for dictionary in dictionaries:
        dictionary['word_count'] = counts.get(dictionary['id'], 0)
        dictionary['can_manage'] = can_manage_dictionary(actor, dictionary)
    return {'dicts': dictionaries, 'pagination': pagination}


def get_dictionary(actor, dict_id):
    with reading() as cursor:
        dictionary = _dictionary(cursor, dict_id)
    dictionary['can_manage'] = can_manage_dictionary(actor, dictionary)
    return dictionary


def create_dictionary(actor_id, name):
    name = string(name, 'Dictionary name', minimum=1)
    with transaction() as cursor:
        lock_actor(cursor, actor_id)
        cursor.execute(
            'INSERT INTO dict (dictname, ownerid, deleted) VALUES (%s, %s, 0)', (name, actor_id)
        )
        dict_id = cursor.lastrowid
    return {'dict_id': dict_id, 'message': 'Dictionary created'}


def update_dictionary(actor_id, dict_id, name):
    name = string(name, 'Dictionary name', minimum=1)
    with transaction() as cursor:
        _writable_dictionary(cursor, actor_id, dict_id)
        cursor.execute('UPDATE dict SET dictname = %s WHERE id = %s', (name, dict_id))
    return {'message': 'Dictionary updated'}


def delete_dictionary(actor_id, dict_id):
    with transaction() as cursor:
        _writable_dictionary(cursor, actor_id, dict_id)
        cursor.execute('UPDATE dict SET deleted = 1 WHERE id = %s', (dict_id,))
        cursor.execute('UPDATE word SET deleted = 1 WHERE dictid = %s', (dict_id,))
    return {'message': 'Dictionary and its words deleted'}


def get_words(dict_id, query=ListQuery()):
    source = 'word'
    where, params = 'dictid = %s AND deleted = 0', [dict_id]
    if query.q:
        where += " AND (english LIKE %s ESCAPE '!' OR chinese LIKE %s ESCAPE '!')"
        params.extend([query.like, query.like])
        # This one-row aggregate exposes concrete scope bounds to MySQL's range
        # optimizer, without forcing an index or adding a network round trip.
        # Unlike scalar subqueries in WHERE, both the primary range and the
        # dictionary range remain useful candidates. Empty bounds match no rows;
        # zero/negative legacy IDs and the original text predicates are preserved.
        source += (
            ' JOIN (SELECT MIN(id) AS first_id, MAX(id) AS last_id FROM word '
            'WHERE dictid = %s AND deleted = 0) bounds '
            'ON id BETWEEN bounds.first_id AND bounds.last_id'
        )
        params.insert(0, dict_id)
    with reading() as cursor:
        _dictionary(cursor, dict_id)
        cursor.execute('SELECT COUNT(*) AS total FROM ' + source + ' WHERE ' + where, params)
        pagination = query.metadata(cursor.fetchone()['total'])
        if pagination['total'] == 0:
            return {'words': [], 'pagination': pagination}
        # Retain the exact search/order while deferring full rows until after the
        # accepted ID page. Contains search may still need text for skipped rows.
        cursor.execute(
            'SELECT w.id, w.dictid, w.english, w.chinese '
            'FROM word w JOIN (SELECT id FROM '
            + source
            + ' WHERE '
            + where
            + ' ORDER BY id LIMIT %s OFFSET %s) page ON page.id = w.id ORDER BY w.id',
            [*params, query.per_page, page_offset(pagination)],
        )
        return {'words': cursor.fetchall(), 'pagination': pagination}


def create_word(actor_id, dict_id, english, chinese):
    english, chinese = string(english, 'English', minimum=1), string(chinese, 'Chinese', minimum=1)
    with transaction() as cursor:
        _writable_dictionary(cursor, actor_id, dict_id)
        cursor.execute(
            'INSERT INTO word (dictid, english, chinese, deleted) VALUES (%s, %s, %s, 0)',
            (dict_id, english, chinese),
        )
        word_id = cursor.lastrowid
    return {'word_id': word_id, 'message': 'Word created'}


def _writable_word(cursor, actor_id, word_id):
    actor = lock_actor(cursor, actor_id)
    cursor.execute('SELECT dictid FROM word WHERE id = %s AND deleted = 0', (word_id,))
    word = cursor.fetchone()
    if not word:
        raise ApiError('Word not found', 404, 'not_found')
    dictionary = _dictionary(cursor, word['dictid'], lock=True)
    require_dictionary_owner(actor, dictionary)
    cursor.execute('SELECT id FROM word WHERE id = %s AND deleted = 0 FOR UPDATE', (word_id,))
    if not cursor.fetchone():
        raise ApiError('Word not found', 404, 'not_found')


def update_word(actor_id, word_id, english, chinese):
    english, chinese = string(english, 'English', minimum=1), string(chinese, 'Chinese', minimum=1)
    with transaction() as cursor:
        _writable_word(cursor, actor_id, word_id)
        cursor.execute(
            'UPDATE word SET english = %s, chinese = %s WHERE id = %s', (english, chinese, word_id)
        )
    return {'message': 'Word updated'}


def delete_word(actor_id, word_id):
    with transaction() as cursor:
        _writable_word(cursor, actor_id, word_id)
        cursor.execute('UPDATE word SET deleted = 1 WHERE id = %s', (word_id,))
    return {'message': 'Word deleted'}


def import_csv(actor_id, dict_id, content):
    rows = parse_csv(
        content,
        max_bytes=current_app.config['CSV_MAX_BYTES'],
        max_rows=current_app.config['CSV_MAX_ROWS'],
    )
    with transaction() as cursor:
        _writable_dictionary(cursor, actor_id, dict_id)
        cursor.executemany(
            'INSERT INTO word (dictid, english, chinese, deleted) VALUES (%s, %s, %s, 0)',
            [(dict_id, english, chinese) for english, chinese in rows],
        )
    return {'count': len(rows), 'message': f'{len(rows)} words imported'}


def export_csv(dict_id):
    with reading() as cursor:
        dictionary = _dictionary(cursor, dict_id)
        cursor.execute(
            'SELECT english, chinese FROM word WHERE dictid = %s AND deleted = 0 ORDER BY id',
            (dict_id,),
        )
        words = cursor.fetchall()
    name = (
        re.sub(r'[\x00-\x1f/\\:*?"<>|]', '_', (dictionary['dictname'] or 'dictionary')).strip(' .')[
            :80
        ]
        or 'dictionary'
    )
    return {'filename': name + '.csv', 'content': csv_content(words)}
