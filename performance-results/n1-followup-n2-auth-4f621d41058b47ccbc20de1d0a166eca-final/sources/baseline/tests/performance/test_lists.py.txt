"""Opt-in latency/plan baseline. Assertions are correctness checks, not noisy time gates."""

import hashlib
from dataclasses import asdict
from functools import partial
from pathlib import Path
from uuid import uuid4

import pytest

from wm_app.services import auth, dictionary, leaderboard, lobby
from wm_app.services.pagination import ListQuery

from . import dataset
from .dataset import DISTRIBUTIONS, seed
from .index_costs import storage, write_costs
from .measurement import environment, measure, save_report, signature

pytestmark = [pytest.mark.mysql, pytest.mark.performance]
ROOT_ACTOR = {'id': 3, 'type': 'root'}


def cases():
    first, last = ListQuery(), ListQuery(page=1000000)
    return [
        ('dictionaries.first', partial(dictionary.list_dictionaries, ROOT_ACTOR, first)),
        ('dictionaries.last', partial(dictionary.list_dictionaries, ROOT_ACTOR, last)),
        (
            'dictionaries.wide',
            partial(dictionary.list_dictionaries, ROOT_ACTOR, ListQuery(per_page=100)),
        ),
        (
            'dictionaries.contains',
            partial(dictionary.list_dictionaries, ROOT_ACTOR, ListQuery(q='needle')),
        ),
        (
            'dictionaries.miss',
            partial(dictionary.list_dictionaries, ROOT_ACTOR, ListQuery(q='no-such-entry')),
        ),
        ('words.first', partial(dictionary.get_words, 10, first)),
        ('words.last', partial(dictionary.get_words, 10, last)),
        ('words.contains', partial(dictionary.get_words, 10, ListQuery(q='needle'))),
        ('words.literal', partial(dictionary.get_words, 10, ListQuery(q='%_!'))),
        ('words.miss', partial(dictionary.get_words, 10, ListQuery(q='no-such-entry'))),
        ('users.first', partial(auth.list_users, ROOT_ACTOR, False, first)),
        ('users.last', partial(auth.list_users, ROOT_ACTOR, False, last)),
        ('users.contains', partial(auth.list_users, ROOT_ACTOR, True, ListQuery(q='needle'))),
        ('users.miss', partial(auth.list_users, ROOT_ACTOR, True, ListQuery(q='no-such-entry'))),
        ('leaderboard.first', partial(leaderboard.list_leaders, first)),
        ('leaderboard.last', partial(leaderboard.list_leaders, last)),
        ('lobby.first', partial(lobby.list_games, 3, first, 'all')),
        ('lobby.last', partial(lobby.list_games, 3, last, 'all')),
        ('lobby.waiting', partial(lobby.list_games, 3, first, 'waiting')),
        ('lobby.contains', partial(lobby.list_games, 3, ListQuery(q='needle'), 'all')),
        ('lobby.miss', partial(lobby.list_games, 3, ListQuery(q='no-such-entry'), 'all')),
        ('lobby.active_member', partial(lobby.list_games, 1, first, 'cancelled')),
        ('small_words.first', partial(dictionary.get_words, 11, first)),
        ('small_words.last', partial(dictionary.get_words, 11, last)),
        ('small_words.contains', partial(dictionary.get_words, 11, ListQuery(q='needle'))),
        ('small_words.literal', partial(dictionary.get_words, 11, ListQuery(q='%_!'))),
        ('small_words.miss', partial(dictionary.get_words, 11, ListQuery(q='no-such-entry'))),
        ('words.common', partial(dictionary.get_words, 10, ListQuery(q='entry'))),
        ('words.chinese', partial(dictionary.get_words, 10, ListQuery(q='词'))),
        (
            'words.contains_last',
            partial(dictionary.get_words, 10, ListQuery(page=1000000, q='needle')),
        ),
        ('users.second', partial(auth.list_users, ROOT_ACTOR, False, ListQuery(page=2))),
        ('users.filtered', partial(auth.list_users, ROOT_ACTOR, False, ListQuery(q='needle'))),
        ('users.literal', partial(auth.list_users, ROOT_ACTOR, False, ListQuery(q='%_!'))),
        ('users.all', partial(auth.list_users, ROOT_ACTOR, True, first)),
    ]


def test_synthetic_list_baseline(perf_app, perf_options):
    path = Path(perf_options['report'] or f'performance-results/lists-{uuid4().hex}.json')
    if path.exists() or path.is_symlink():
        pytest.fail('Report path already exists; choose a new path')
    counts = seed(perf_app, perf_options['size'], perf_options['distribution'])
    profile = DISTRIBUTIONS[perf_options['distribution']]
    results = {}
    for name, operation in cases():
        results[name] = measure(perf_app, operation, perf_options['runs'])
        print(f'{name}: median {results[name]["service_ms"]["median"]:.2f} ms', flush=True)
    assert results['dictionaries.first']['pagination']['total'] == 1 + sum(
        i == 0 or i % 23 != 0 for i in range(counts['dictionaries'])
    )
    assert results['words.first']['pagination']['total'] == sum(
        profile.dictionary_id(i, counts) == 10 and profile.deleted(i) == 0
        for i in range(counts['words'])
    )
    assert results['users.first']['pagination']['total'] == 3 + sum(
        profile.deleted(i) == 0 for i in range(counts['users'])
    )
    assert results['leaderboard.first']['pagination']['total'] == min(
        100, results['users.first']['pagination']['total']
    )
    assert results['lobby.first']['pagination']['total'] == counts['games'] + 1
    assert results['lobby.first']['active_game_id'] is None
    assert (
        results['lobby.active_member']['active_game_id']
        == 100000 + (counts['games'] - 1) // 20 * 20
    )
    assert all(
        result['pagination']['total'] == 0
        for name, result in results.items()
        if name.endswith('.miss')
    )
    report = {
        'format_version': 1,
        'scope': 'synthetic sequential service reads; not production acceptance or HTTP load',
        'profile': perf_options['size'],
        'distribution': {'name': perf_options['distribution'], **asdict(profile)},
        'dataset_sha256': hashlib.sha256(Path(dataset.__file__).read_bytes()).hexdigest(),
        'generated_rows': counts,
        'warmups_per_case': 1,
        'measured_runs_per_case': perf_options['runs'],
        'environment': environment(perf_app),
        'storage': storage(perf_app),
        'cases': results,
    }
    if perf_options['writes']:
        report['write_costs'] = write_costs(perf_app, perf_options['runs'])
        assert environment(perf_app)['row_counts'] == report['environment']['row_counts']
        with perf_app.app_context():
            for name, operation in cases():
                assert signature(operation()) == results[name]['result_sha256']
        report['write_costs']['post_probe_result_checks'] = len(results)
    save_report(path, report)
    print(f'Performance report: {path.resolve()}', flush=True)
